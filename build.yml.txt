name: Build Windows EXE

on:
  push:
    branches: [ "main", "master" ]
  workflow_dispatch:

jobs:
  build:
    runs-on: windows-latest
    steps:
    - uses: actions/checkout@v3

    - name: Set up Python
      uses: actions/setup-python@v4
      with:
        python-version: '3.10'

    - name: Install dependencies & Build EXE
      run: |
        python -m pip install --upgrade pip
        pip install tkinterdnd2 requests pyinstaller
        pyinstaller --noconsole --onefile --collect-data tkinterdnd2 --name="FIBA原圖下載器" fiba_downloader.py

    - name: Upload EXE
      uses: actions/upload-artifact@v3
      with:
        name: FIBA原圖下載器_EXE
        path: dist/FIBA原圖下載器.exe
